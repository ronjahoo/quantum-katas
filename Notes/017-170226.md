```
from pytest import approx

@exercise
def is_matrix_unitary(a : Matrix) -> bool:
    adj = adjoint(a)
    c = create_empty_matrix(len(a), len(a))

    for i in range(len(a)):
        for j in range(len(a)):
            c[i][j] = 0
            for k in range(len(a)):
                c[i][j] += a[i][k] * adj[k][j]

    for i in range(len(a)):
        for j in range(len(a)):
            if i == j:
                if c[i][j] != 1:
                    return False
            else:
                if c[i][j] != 0:
                    return False
    return True
```
