def polar_mult(x : Polar, y : Polar) -> Polar:
    radius = x[0] * y[0]
    angle = x[1] + y[1]
    
    if (angle > math.pi):
        angle = angle - 2.0 * math.pi
        
    elif (angle <= -math.pi):
        angle = angle + 2.0 * math.pi
    
    return (radius, angle)
