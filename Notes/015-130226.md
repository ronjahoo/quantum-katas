```
def conjugate(a : Matrix) -> Matrix:
    b = create_empty_matrix(len(a), len(a[0]))
    for i in range(len(a)):
        for j in range(len(a[0])):
            z = a[i][j]
            b[i][j] = (complex(z.real, -z.imag))
    return b
```
