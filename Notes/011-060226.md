```
def scalar_mult(x : complex, a : Matrix) -> Matrix:
    rows = len(a)
    columns = len(a[0])
    
    c = create_empty_matrix(rows, columns)
    
    for i in range(rows):
        for j in range(columns):
            y = a[i][j]
            
            c[i][j] = y*x
    
    return c
```
