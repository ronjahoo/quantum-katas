```
@exercise
def matrix_mult(a : Matrix, b : Matrix) -> Matrix:
    c = create_empty_matrix(len(a), len(b[0]))
    for i in range(len(a)):
        for j in range(len(b[0])):
            sum = 0
            for n in range(len(b)):
                sum = sum + a[i][n] * b[n][j]
                c[i][j] = sum
    return c
```
