@exercise
def find_eigenvector(a : Matrix, x : float) -> Matrix:
    if (a[0][1] == 0):
        if (a[0][0] - x == 0):
            if (a[1][0] == 0):
                return [[1], [0]]
            else:
                return [[(a[1][1] - x) / (-a[1][0])], [1]]
        else:
            return [[0], [1]]
    
    v0 = 1
    v1 = (a[0][0] - x) / (-a[0][1])
    return [[v0], [v1]]
