def complex_exp_arbitrary(x : Complex, y : Complex) -> Complex:
    r = math.sqrt(x[0] ** 2 + x[1] ** 2)
    theta = math.atan2(x[1], x[0])
    
    if (r == 0):
        return (0, 0)

    lnr = math.log(r)
    exp = math.exp(lnr * y[0] - y[1] * theta)
    
    r = exp * (math.cos(lnr * y[1] + theta * y[0] ))
    i = exp * (math.sin(lnr * y[1] + theta * y[0]))
    return (r, i)
