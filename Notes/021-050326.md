```
def tensor_product(a : Matrix, b : Matrix) -> Matrix:
    aRows = len(a)       
    aColumns = len(a[0]) 
    bRows = len(b)       
    bColumns = len(b[0]) 
    
    c = create_empty_matrix(aRows * bRows, aColumns *  bColumns)
    
    for i in range(aRows):
        for j in range(aColumns):
            for k in range(bRows):
                for l in range(bColumns):
                    c[i * bRows + k][j * bColumns + l] = a[i][j] * b[k][l]
    
    return c
```
