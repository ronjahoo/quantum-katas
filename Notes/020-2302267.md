def outer_prod(v : Matrix, w : Matrix) -> Matrix:
    wC = adjoint(w)
    c = create_empty_matrix(len(v), len(w))
    for i in range(len(v)):
        for j in range(len(w)):
            c[i][j] = v[i][0] * wC[0][j]
    return c
