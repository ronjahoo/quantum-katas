def normalize(v : Matrix) -> Matrix:
    norm = math.sqrt(inner_prod(v, v).real)
    
    n = len(v)
    result = create_empty_matrix(n, 1)
    
    for i in range(n):
        result[i][0] = v[i][0] / norm
        
    return result
